+def select(parcel):
+    """Pick the cheapest service; ties go to the alphabetically first name."""
+    options = quotes(parcel)
+    if not options:
+        return {"id": parcel["id"], "status": "unavailable",
+                "reason": f"no service available for zone {parcel['zone']!r}"}
+    best = min(options, key=lambda q: (q["cost"], q["service"]))
+    return {"id": parcel["id"], "status": "selected", "service": best["service"],
+            "days": best["days"], "cost": best["cost"]}
+
+
+def select_batch(parcels):
+    return {"parcels": [select(p) for p in parcels]}
