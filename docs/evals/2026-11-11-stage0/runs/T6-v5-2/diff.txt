+"""Build the invoice export (docs/accounting.md): per-line rounding, signed credits, no tax."""
+import argparse
+import json
+from itertools import groupby
+from pathlib import Path
+
+from invoice import line_total, load, total
+
+
+def fmt(amount):
+    """Exactly two decimal places, trailing zeros kept."""
+    return format(amount, "f")
+
+
+def build_export(rows):
+    # sorted() is stable, so source line order is preserved within each invoice.
+    ordered = sorted(rows, key=lambda r: r["invoice"])
+    invoices = []
+    for invoice_id, group in groupby(ordered, key=lambda r: r["invoice"]):
+        group = list(group)
+        lines = []
+        for r in group:
+            line = {"item": r["item"], "amount": fmt(line_total(r))}
+            if r.get("tax_class"):
+                line["tax_class"] = r["tax_class"]
+            lines.append(line)
+        invoices.append({"id": invoice_id, "total": fmt(total(group)), "lines": lines})
+    return {"invoices": invoices}
+
+
+def write_export(source, destination):
+    destination = Path(destination)
+    destination.parent.mkdir(parents=True, exist_ok=True)
+    payload = build_export(load(source))
+    with destination.open("w", encoding="utf-8") as stream:
+        json.dump(payload, stream, indent=2, sort_keys=True)
+        stream.write("\n")
+    return payload
+
+
+if __name__ == "__main__":
+    parser = argparse.ArgumentParser(description="Write the invoice batch export.")
+    parser.add_argument("--source", default="data/lines.csv")
+    parser.add_argument("--out", default="build/invoices.json")
+    args = parser.parse_args()
+    write_export(args.source, args.out)
+    print(f"wrote {args.out}")

+import json
+import subprocess
+import sys
+from decimal import Decimal
+
+from export import build_export
+from invoice import load, total
+
+
+def row(inv, item, qty, price, tax="standard"):
+    return {"invoice": inv, "item": item, "quantity": qty, "unit_price": price, "tax_class": tax}
+
+
+def test_per_line_rounding_not_single_rounding():
+    rows = [row("I1", f"U{i}", "1", "0.335") for i in range(3)]
+    inv = build_export(rows)["invoices"][0]
+    assert inv["total"] == "1.02"
+    assert [line["amount"] for line in inv["lines"]] == ["0.34"] * 3
+
+
+def test_fixture_batch_total_and_refund_sign():
+    inv = build_export(load("data/lines.csv"))["invoices"][0]
+    assert inv["id"] == "I100"
+    assert inv["total"] == "0.92"
+    assert inv["lines"][-1] == {"item": "Refund", "amount": "-0.10", "tax_class": "exempt"}
+
+
+def test_negative_half_cent_rounds_away_from_zero():
+    inv = build_export([row("I1", "Credit", "-1", "0.335")])["invoices"][0]
+    assert inv["lines"][0]["amount"] == "-0.34"
+    assert inv["total"] == "-0.34"
+
+
+def test_all_credit_invoice_total_negative():
+    inv = build_export([row("I1", "A", "-2", "1.50"), row("I1", "B", "-1", "0.25")])["invoices"][0]
+    assert inv["total"] == "-3.25"
+
+
+def test_trailing_zeros_and_string_type():
+    inv = build_export([row("I1", "A", "1", "1"), row("I1", "B", "1", "0.9")])["invoices"][0]
+    assert [line["amount"] for line in inv["lines"]] == ["1.00", "0.90"]
+    assert inv["total"] == "1.90"
+    assert all(isinstance(line["amount"], str) for line in inv["lines"])
+    assert isinstance(inv["total"], str)
+
+
+def test_zero_total_has_two_decimals():
+    inv = build_export([row("I1", "A", "1", "1.00"), row("I1", "B", "-1", "1.00")])["invoices"][0]
+    assert inv["total"] == "0.00"
+
+
+def test_grouping_order_and_sorting():
+    rows = [row("I2", "b1", "1", "1"), row("I1", "a1", "1", "2"), row("I2", "b2", "1", "3"), row("I1", "a2", "1", "4")]
+    invs = build_export(rows)["invoices"]
+    assert [i["id"] for i in invs] == ["I1", "I2"]
+    assert [l["item"] for l in invs[0]["lines"]] == ["a1", "a2"]
+    assert [l["item"] for l in invs[1]["lines"]] == ["b1", "b2"]
+
+
+def test_totals_agree_with_invoice_total_and_no_tax_added():
+    rows = load("data/lines.csv")
+    inv = build_export(rows)["invoices"][0]
+    assert Decimal(inv["total"]) == total(rows)
+    assert isinstance(total(rows), Decimal)
+    assert set(inv) == {"id", "total", "lines"}
+    assert all(set(line) <= {"item", "amount", "tax_class"} for line in inv["lines"])
+
+
+def test_command_writes_artifact(tmp_path):
+    out = tmp_path / "nested" / "invoices.json"
+    subprocess.run([sys.executable, "export.py", "--out", str(out)], check=True)
+    data = json.loads(out.read_text())
+    assert data["invoices"][0]["total"] == "0.92"
