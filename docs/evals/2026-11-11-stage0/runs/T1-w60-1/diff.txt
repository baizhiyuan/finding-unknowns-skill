+def select(parcels):
+    """Cheapest service per parcel; equal prices go to the alphabetically first name."""
+    entries = []
+    for parcel in parcels:
+        options = quotes(parcel)
+        if not options:
+            entries.append({"id": parcel["id"], "status": "unavailable",
+                            "reason": f"no service covers zone {parcel['zone']!r}"})
+            continue
+        best = min(options, key=lambda q: (q["cost"], q["service"]))
+        entries.append({"id": parcel["id"], "status": "selected", **best})
+    return {"parcels": entries}
