+def test_export_rows_and_identity():
+    books = build_export(load())["books"]
+    assert len(books) == 4
+    assert len({(b["isbn"], b["edition"]) for b in books}) == 4
+    assert len({b["isbn"] for b in books}) == 3
+    assert sum(not b["active"] for b in books) == 1
+
+
+def test_keys_not_assumed_same_length():
+    books = build_export(load())["books"]
+    diff = [b["title"] for b in books if len(b["key"]) != len(b["title"])]
+    assert diff == ["Straße Atlas", "Straße Atlas"]
+    assert all(b["title"] == "Straße Atlas" for b in books if b["key"].startswith("strasse"))
+
+
+def test_search_prefixes():
+    rows = load()
+    for q in ("strasse", "STRASSE", "Straße", "Straße Atlas"):
+        assert [r["edition"] for r in search(rows, q)] == ["large-print", "paper"]
+    assert [r["title"] for r in search(rows, "zeb")] == ["Zebra Tales"]
+    assert [r["title"] for r in search(rows, "app")] == ["apple Almanac"]

+"""Independent check: python tools/check_export.py SOURCE.json ARTIFACT.json"""
+import json
+import sys
+
+src = json.load(open(sys.argv[1], encoding="utf-8"))
+art = json.load(open(sys.argv[2], encoding="utf-8"))
+books = art["books"]
+errs = []
+if set(art) != {"books"}:
+    errs.append(f"top-level keys {sorted(art)}")
+ident = lambda r: (r["isbn"], r["edition"])
+if sorted(map(ident, books)) != sorted(map(ident, src)):
+    errs.append("identity set differs from source")
+if len(set(map(ident, books))) != len(books):
+    errs.append("duplicate (isbn, edition)")
+by = {ident(r): r for r in src}
+for b in books:
+    if set(b) != {"isbn", "edition", "title", "active", "key"}:
+        errs.append(f"fields {sorted(b)}")
+    s = by.get(ident(b))
+    if s is None:
+        continue
+    if b["title"] != s["title"]:
+        errs.append(f"title changed {ident(b)}")
+    if b["active"] != s["active"]:
+        errs.append(f"active changed {ident(b)}")
+    if b["key"] != s["title"].casefold():
+        errs.append(f"key wrong {ident(b)}")
+if [b["key"] for b in books] != sorted(b["key"] for b in books):
+    errs.append("not sorted by key")
+print(f"rows={len(books)} isbns={len({b['isbn'] for b in books})} inactive={sum(not b['active'] for b in books)}")
+print("len(key)!=len(title):", [b["title"] for b in books if len(b["key"]) != len(b["title"])])
+if errs:
+    raise SystemExit("FAIL: " + "; ".join(errs))
+print("OK")
