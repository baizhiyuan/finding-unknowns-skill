+def build_export(rows):
+    """Search export rows: all stock (inactive included), identity (isbn, edition),
+    ordered by displayed title (Python str order), then isbn, then edition."""
+    seen = set()
+    books = []
+    for r in rows:
+        ident = (r["isbn"], r["edition"])
+        if ident in seen:
+            raise ValueError(f"duplicate row identity: {ident}")
+        seen.add(ident)
+        books.append({"isbn": r["isbn"], "edition": r["edition"], "title": r["title"],
+                      "active": r["active"], "key": search_key(r["title"])})
+    books.sort(key=lambda b: (b["title"], b["isbn"], b["edition"]))
+    return {"books": books}
+
+
+def write_export(path="build/search.json", source="data/books.json"):
+    path = Path(path)
+    path.parent.mkdir(parents=True, exist_ok=True)
+    path.write_text(json.dumps(build_export(load(source)), ensure_ascii=False, indent=2) + "\n",
+                    encoding="utf-8")
