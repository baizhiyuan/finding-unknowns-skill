+def build_export(rows):
+    """Search export: one entry per (isbn, edition), inactive stock kept (flagged
+    via "active"), ordered by key then identity so output is deterministic and
+    independent of storage order."""
+    seen = set()
+    books = []
+    for r in rows:
+        ident = (r["isbn"], r["edition"])
+        if ident in seen:
+            raise ValueError(f"duplicate row identity {ident}")
+        seen.add(ident)
+        books.append({"isbn": r["isbn"], "edition": r["edition"], "title": r["title"],
+                      "active": r["active"], "key": search_key(r["title"])})
+    books.sort(key=lambda b: (b["key"], b["isbn"], b["edition"]))
+    return {"books": books}
+
+
+def write_export(out, rows=None):
+    data = build_export(load() if rows is None else rows)
+    out = Path(out)
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(data, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
+                   encoding="utf-8")
