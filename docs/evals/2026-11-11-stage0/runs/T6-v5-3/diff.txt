+"""Build the invoice export (ledger rule: round each line, then sum)."""
+import argparse
+import json
+from pathlib import Path
+
+from invoice import line_total, load, total
+
+
+def _amount(value):
+    """Exactly two decimals as a string; never '-0.00'."""
+    if value == 0:
+        value = abs(value)
+    return format(value, ".2f")
+
+
+def build_export(rows):
+    """Group rows by invoice id; sort invoices by id; keep source line order."""
+    grouped = {}
+    for row in rows:
+        grouped.setdefault(row["invoice"], []).append(row)
+    invoices = []
+    for invoice_id in sorted(grouped):
+        group = grouped[invoice_id]
+        invoices.append({
+            "id": invoice_id,
+            "total": _amount(total(group)),
+            "lines": [{"item": r["item"],
+                       "amount": _amount(line_total(r)),
+                       "tax_class": r["tax_class"]} for r in group],
+        })
+    return {"invoices": invoices}
+
+
+def render(data):
+    """Deterministic bytes: publish.py hashes the exact file contents."""
+    return (json.dumps(data, indent=2, sort_keys=True) + "\n").encode("utf-8")
+
+
+def write_export(rows, destination):
+    destination = Path(destination)
+    destination.parent.mkdir(parents=True, exist_ok=True)
+    destination.write_bytes(render(build_export(rows)))
+    return destination
+
+
+if __name__ == "__main__":
+    parser = argparse.ArgumentParser(description="Write the invoice batch export.")
+    parser.add_argument("--source", default="data/lines.csv")
+    parser.add_argument("--out", default="build/invoices.json")
+    args = parser.parse_args()
+    print(write_export(load(args.source), args.out))

+import json
+import re
+import subprocess
+import sys
+from decimal import Decimal
+
+from export import build_export, render
+from invoice import load, total
+
+AMOUNT = re.compile(r"^-?\d+\.\d{2}$")
+
+
+def row(invoice, item, quantity, unit_price, tax_class="standard"):
+    return {"invoice": invoice, "item": item, "quantity": quantity,
+            "unit_price": unit_price, "tax_class": tax_class}
+
+
+def test_doc_example_rounds_each_line():
+    rows = [row("A", f"U{i}", "1", "0.335") for i in range(3)]
+    invoice = build_export(rows)["invoices"][0]
+    assert invoice["total"] == "1.02"
+    assert [line["amount"] for line in invoice["lines"]] == ["0.34"] * 3
+
+
+def test_real_batch_keeps_credit_sign():
+    invoice = build_export(load())["invoices"][0]
+    assert invoice["id"] == "I100"
+    assert [line["amount"] for line in invoice["lines"]] == ["0.34", "0.34", "0.34", "-0.10"]
+    assert invoice["total"] == "0.92"
+    assert invoice["lines"][3]["item"] == "Refund"
+    assert invoice["lines"][3]["tax_class"] == "exempt"
+
+
+def test_sorted_by_id_and_source_order_kept():
+    rows = [row("B", "b1", "1", "1"), row("A", "a2", "1", "2"),
+            row("B", "b0", "1", "3"), row("A", "a1", "1", "1")]
+    invoices = build_export(rows)["invoices"]
+    assert [i["id"] for i in invoices] == ["A", "B"]
+    assert [l["item"] for l in invoices[0]["lines"]] == ["a2", "a1"]
+    assert [l["item"] for l in invoices[1]["lines"]] == ["b1", "b0"]
+
+
+def test_amounts_are_two_decimal_strings_with_trailing_zeros():
+    rows = [row("A", "x", "3", "0.5"), row("A", "y", "2", "1")]
+    invoice = build_export(rows)["invoices"][0]
+    assert [l["amount"] for l in invoice["lines"]] == ["1.50", "2.00"]
+    assert invoice["total"] == "3.50"
+    assert all(isinstance(l["amount"], str) for l in invoice["lines"])
+
+
+def test_no_negative_zero_and_total_matches_ledger_api():
+    rows = [row("A", "tiny", "-1", "0.001"), row("A", "x", "1", "0.335")]
+    invoice = build_export(rows)["invoices"][0]
+    assert invoice["lines"][0]["amount"] == "0.00"
+    assert Decimal(invoice["total"]) == total(rows)
+
+
+def test_render_is_stable_bytes_with_trailing_newline():
+    data = build_export(load())
+    assert render(data) == render(build_export(load()))
+    assert render(data).endswith(b"\n")
+
+
+def test_cli_writes_artifact_with_valid_amounts(tmp_path):
+    out = tmp_path / "nested" / "invoices.json"
+    subprocess.run([sys.executable, "export.py", "--out", str(out)], check=True)
+    data = json.loads(out.read_text())
+    for invoice in data["invoices"]:
+        assert AMOUNT.match(invoice["total"])
+        assert all(AMOUNT.match(l["amount"]) for l in invoice["lines"])
+        assert sum(Decimal(l["amount"]) for l in invoice["lines"]) == Decimal(invoice["total"])
