+"""Contract checks for the search export. Written before the build; not edited by the implementer."""
+import hashlib
+import json
+import catalog
+
+EXPECTED = [
+    {"isbn": "9780000000011", "edition": "large-print", "title": "Straße Atlas", "active": True, "key": "strasse atlas"},
+    {"isbn": "9780000000011", "edition": "paper", "title": "Straße Atlas", "active": True, "key": "strasse atlas"},
+    {"isbn": "9780000000028", "edition": "paper", "title": "Zebra Tales", "active": False, "key": "zebra tales"},
+    {"isbn": "9780000000035", "edition": "paper", "title": "apple Almanac", "active": True, "key": "apple almanac"},
+]
+
+
+def test_export_exact():
+    assert catalog.build_export() == {"books": EXPECTED}
+
+
+def test_key_lengths_differ_only_for_strasse():
+    rows = catalog.build_export()["books"]
+    assert [r["isbn"] for r in rows if len(r["key"]) != len(r["title"])] == ["9780000000011"] * 2
+
+
+def test_build_is_byte_identical(tmp_path):
+    a, b = tmp_path / "a.json", tmp_path / "b.json"
+    catalog.write_export(a)
+    catalog.write_export(b)
+    assert hashlib.sha256(a.read_bytes()).digest() == hashlib.sha256(b.read_bytes()).digest()
+    assert json.loads(a.read_text(encoding="utf-8")) == {"books": EXPECTED}
+
+
+def _ids(prefix):
+    return [(r["isbn"], r["edition"]) for r in catalog.search(prefix)]
+
+
+def test_search_sets():
+    assert _ids("strass") == [("9780000000011", "large-print"), ("9780000000011", "paper")]
+    assert _ids("STRAß") == _ids("strass")
+    assert _ids("zeb") == [("9780000000028", "paper")]
+    assert _ids("APP") == [("9780000000035", "paper")]
+    assert _ids("x") == []
