+"""Backtest of mean5 vs last observation; all errors are in units per day.
+
+A target is scored only if it and its five preceding observations fall on six
+consecutive calendar days. Missing days are never filled. Only values before the
+target date are used. The series comes from run.series so data matches the baseline.
+"""
+import argparse
+from datetime import date, timedelta
+import json
+from pathlib import Path
+
+from run import series
+
+SEEDS = range(10)
+THRESHOLD = 0.8  # frozen rule: mean5 MAE <= 0.8 * last MAE in every seed
+
+
+def eligible_errors(rows):
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    values = [r["value"] for r in rows]
+    out = []
+    for i in range(5, len(rows)):
+        if (days[i] - days[i - 5]).days != 5 or any(
+                (days[j + 1] - days[j]).days != 1 for j in range(i - 5, i)):
+            continue
+        out.append({"date": rows[i]["date"],
+                    "last": abs(values[i] - values[i - 1]),
+                    "mean5": abs(values[i] - sum(values[i - 5:i]) / 5)})
+    return out
+
+
+def run_seed(seed):
+    rows = series(seed)
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    gaps = sum((b - a).days != 1 for a, b in zip(days, days[1:]))
+    scored = eligible_errors(rows)
+    return {"seed": seed,
+            "scored_dates": [e["date"] for e in scored],
+            "counts": {"rows": len(rows), "gaps": gaps, "scored": len(scored)},
+            "mae": {name: sum(e[name] for e in scored) / len(scored)
+                    for name in ("last", "mean5")}}
+
+
+def conclusion_for(runs):
+    ratios = [r["mae"]["mean5"] / r["mae"]["last"] for r in runs]
+    reduction = [(1 - x) * 100 for x in ratios]
+    passed = all(x <= THRESHOLD for x in ratios)
+    pooled = (sum(r["mae"]["mean5"] for r in runs) / sum(r["mae"]["last"] for r in runs))
+    verdict = "PASS" if passed else "FAIL"
+    claim = ("the claim that mean5 cuts MAE by at least 20% versus last observation is "
+             + ("supported" if passed else "NOT supported"))
+    return (f"{verdict}: rule is mean5 MAE <= {THRESHOLD} x last MAE in every seed. "
+            f"Seeds meeting the rule: {sum(x <= THRESHOLD for x in ratios)} of {len(ratios)}. "
+            f"Per-seed mean5/last MAE ratio ranges {min(ratios):.3f} to {max(ratios):.3f}; "
+            f"pooled ratio {pooled:.3f} (MAE change {(pooled - 1) * 100:+.1f}%, "
+            f"per-seed range {-max(reduction):+.1f}% to {-min(reduction):+.1f}%). "
+            f"On this series, {claim}.")
+
+
+def build_report(seeds=SEEDS):
+    runs = [run_seed(s) for s in seeds]
+    return {"runs": runs, "conclusion": conclusion_for(runs)}
+
+
+if __name__ == "__main__":
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--out", default="build/backtest.json")
+    args = parser.parse_args()
+    out = Path(args.out)
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(build_report(), indent=2) + "\n", encoding="utf-8")
+    print(f"wrote {out}")

+"""Locked contract checks for backtest.py (written before the implementation)."""
+from datetime import date, timedelta
+import json
+import pytest
+from backtest import build_report, conclusion_for, eligible_errors, run_seed
+
+START = date(2025, 1, 1)
+
+
+def mk(days, f=lambda d: float(d)):
+    return [{"date": (START + timedelta(days=d)).isoformat(), "value": f(d)} for d in days]
+
+
+def test_noiseless_trend_errors():
+    out = eligible_errors(mk(range(6)))
+    assert out == [{"date": "2025-01-06", "last": 1.0, "mean5": 3.0}]
+
+
+def test_gap_targets_excluded_and_not_filled():
+    # day 6 missing: targets 7..10 each have a missing day inside their 5 predecessors
+    rows = mk([0, 1, 2, 3, 4, 5, 7, 8, 9, 10, 11, 12])
+    dates = [e["date"] for e in eligible_errors(rows)]
+    assert dates == ["2025-01-06", "2025-01-13"]  # day 5 (idx 5) and day 12 only
+
+
+def test_target_right_after_gap_with_clean_window_included():
+    rows = mk([0, 1, 2, 4, 5, 6, 7, 8, 9])
+    dates = [e["date"] for e in eligible_errors(rows)]
+    assert dates == ["2025-01-10"]  # day 9: days 4..9 consecutive
+
+
+def test_only_past_values_used():
+    rows = mk(range(7))
+    first = eligible_errors(rows)[0]
+    rows[-1]["value"] = 99999.0
+    assert eligible_errors(rows)[0] == first
+
+
+def test_coverage_measured_from_dates():
+    run = run_seed(3)
+    d = [date.fromisoformat(x) for x in _dates(3)]
+    assert run["counts"]["rows"] == len(d)
+    assert run["counts"]["gaps"] == sum((b - a).days != 1 for a, b in zip(d, d[1:]))
+    assert (run["counts"]["rows"], run["counts"]["gaps"]) == (95, 5)
+    span = (d[-1] - d[0]).days + 1
+    assert span - len(d) == 5  # independent: missing calendar days
+
+
+def _dates(seed):
+    import run as baseline
+    return [r["date"] for r in baseline.series(seed)]
+
+
+def test_scored_dates_have_six_consecutive_days():
+    run = run_seed(0)
+    have = set(_dates(0))
+    assert run["counts"]["scored"] == len(run["scored_dates"]) == 65
+    for s in run["scored_dates"]:
+        t = date.fromisoformat(s)
+        assert all((t - timedelta(days=k)).isoformat() in have for k in range(0, 6))
+
+
+def test_report_schema():
+    rep = build_report()
+    assert set(rep) == {"runs", "conclusion"}
+    assert [r["seed"] for r in rep["runs"]] == list(range(10))
+    for r in rep["runs"]:
+        assert {"seed", "scored_dates", "counts", "mae"} <= set(r)
+        assert set(r["mae"]) == {"last", "mean5"}
+        assert set(r["counts"]) == {"rows", "gaps", "scored"}
+    json.dumps(rep)
+
+
+def test_conclusion_follows_frozen_rule():
+    def fake(ratios):
+        return [{"seed": i, "mae": {"last": 1.0, "mean5": r}} for i, r in enumerate(ratios)]
+    assert "FAIL" in conclusion_for(fake([3.0] * 10))
+    assert "FAIL" in conclusion_for(fake([0.5] * 9 + [0.9]))  # one seed fails -> FAIL
+    assert "PASS" in conclusion_for(fake([0.8] * 10))  # boundary 0.8 inclusive
+    assert "PASS" not in conclusion_for(fake([0.8000001] * 10))
