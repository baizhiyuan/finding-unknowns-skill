+import json
+import subprocess
+import sys
+from pathlib import Path
+
+from catalog import build_export, load, search
+
+ROOT = Path(__file__).resolve().parent.parent
+
+
+def export():
+    return build_export(load(str(ROOT / "data" / "books.json")))
+
+
+def ids(books):
+    return [(b["isbn"], b["edition"]) for b in books]
+
+
+def test_one_row_per_isbn_edition_none_merged():
+    books = export()["books"]
+    source = {(r["isbn"], r["edition"]) for r in load(str(ROOT / "data" / "books.json"))}
+    assert len(books) == 4
+    assert len({b["isbn"] for b in books}) == 3
+    assert set(ids(books)) == source and len(set(ids(books))) == 4
+
+
+def test_inactive_stock_is_included_with_flag():
+    by_id = {(b["isbn"], b["edition"]): b for b in export()["books"]}
+    assert by_id[("9780000000028", "paper")]["active"] is False
+    assert by_id[("9780000000011", "paper")]["active"] is True
+
+
+def test_title_untouched_and_key_is_casefold():
+    src = {(r["isbn"], r["edition"]): r for r in load(str(ROOT / "data" / "books.json"))}
+    for b in export()["books"]:
+        assert b["title"] == src[(b["isbn"], b["edition"])]["title"]
+        assert b["key"] == b["title"].casefold()
+    strasse = [b for b in export()["books"] if b["title"] == "Straße Atlas"]
+    assert len(strasse) == 2
+    assert all(len(b["key"]) == 13 and len(b["title"]) == 12 for b in strasse)
+
+
+def test_schema():
+    data = export()
+    assert list(data) == ["books"]
+    for b in data["books"]:
+        assert set(b) == {"isbn", "edition", "title", "active", "key"}
+
+
+def test_order_is_deterministic_by_key_isbn_edition():
+    books = export()["books"]
+    assert books == sorted(books, key=lambda b: (b["key"], b["isbn"], b["edition"]))
+
+
+def test_prefix_search():
+    data = export()
+    for q in ("strasse", "STRASSE", "straß", "Straße Atl", "stras"):
+        assert sorted(ids(search(data, q))) == [
+            ("9780000000011", "large-print"), ("9780000000011", "paper")], q
+    assert ids(search(data, "app")) == [("9780000000035", "paper")]
+    assert ids(search(data, "APP")) == [("9780000000035", "paper")]
+    assert ids(search(data, "zeb")) == [("9780000000028", "paper")]  # inactive is findable
+    assert search(data, "zz") == []
+    assert search(data, "tlas") == []  # prefix, not substring
+    # results show the original spelling, not the key
+    assert {b["title"] for b in search(data, "strasse")} == {"Straße Atlas"}
+
+
+def test_export_command_is_deterministic_and_valid(tmp_path):
+    outs = []
+    for name in ("a.json", "b.json"):
+        out = tmp_path / "sub" / name
+        subprocess.run([sys.executable, "catalog.py", "export", str(out)],
+                       cwd=ROOT, check=True, capture_output=True)
+        outs.append(out.read_bytes())
+    assert outs[0] == outs[1]
+    assert json.loads(outs[0]) == export()
+    assert "Straße".encode("utf-8") in outs[0]  # not \u-escaped away
+
+
+def test_search_command(tmp_path):
+    out = subprocess.run([sys.executable, "catalog.py", "search", "STRASSE"],
+                         cwd=ROOT, check=True, capture_output=True, text=True).stdout
+    assert out.count("Straße Atlas") == 2 and "Zebra" not in out
