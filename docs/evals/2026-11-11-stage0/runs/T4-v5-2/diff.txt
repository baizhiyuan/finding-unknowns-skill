+"""Locked check for build/backtest.json. Usage: python checks/verify_report.py ARTIFACT.json
+Independent recomputation from calendar dates; exits nonzero on any failure."""
+import copy, json, subprocess, sys, tempfile
+from datetime import date, timedelta
+from pathlib import Path
+
+sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
+from run import series  # shared generator (documented limitation)
+
+SEEDS = range(10)
+fails = []
+
+
+def check(ok, msg):
+    if not ok:
+        fails.append(msg)
+    print(("ok   " if ok else "FAIL ") + msg)
+
+
+def recompute(seed):
+    rows = series(seed)
+    d = {date.fromisoformat(r["date"]): r["value"] for r in rows}
+    days = sorted(d)
+    first, last_day = days[0], days[-1]
+    gaps = sum(1 for k in range((last_day - first).days + 1)
+               if first + timedelta(days=k) not in d)
+    out = {"rows": len(rows), "calendar_days": (last_day - first).days + 1, "gaps": gaps}
+    prim = []  # observed-value windows
+    for i in range(5, len(days)):
+        t = days[i]
+        prim.append((t, abs(d[t] - d[days[i - 1]]),
+                     abs(d[t] - sum(d[x] for x in days[i - 5:i]) / 5)))
+    sens = []  # all 5 preceding calendar days observed
+    for t in days:
+        prev = [t - timedelta(days=k) for k in range(1, 6)]
+        if all(p in d for p in prev):
+            sens.append((t, abs(d[t] - d[prev[0]]), abs(d[t] - sum(d[p] for p in prev) / 5)))
+    out["primary"], out["sens"] = prim, sens
+    return out
+
+
+def mae(t):
+    return sum(x[1] for x in t) / len(t), sum(x[2] for x in t) / len(t)
+
+
+art_path = sys.argv[1]
+raw = Path(art_path).read_bytes()
+art = json.loads(raw)
+check("runs" in art and "conclusion" in art, "top-level fields runs and conclusion present")
+runs = art["runs"]
+check([r["seed"] for r in runs] == list(SEEDS), "runs cover seeds 0..9 in order, once each")
+
+P = {"last": [], "mean5": []}
+S = {"last": [], "mean5": []}
+for r in runs:
+    c = recompute(r["seed"])
+    s = r["seed"]
+    check(r["rows"] == c["rows"] and r["gaps"] == c["gaps"] and r["calendar_days"] == c["calendar_days"],
+          f"seed {s}: rows/gaps/calendar_days match dates ({c['rows']}/{c['gaps']}/{c['calendar_days']})")
+    check(r["scored_dates"] == [t.isoformat() for t, _, _ in c["primary"]] and r["scored"] == len(c["primary"]),
+          f"seed {s}: primary scored dates and count match")
+    ml, m5 = mae(c["primary"])
+    check(abs(r["mae"]["last"] - ml) < 1e-9 and abs(r["mae"]["mean5"] - m5) < 1e-9,
+          f"seed {s}: primary MAEs match recomputation")
+    sv = r["sensitivity"]
+    sl, s5 = mae(c["sens"])
+    check(sv["scored_dates"] == [t.isoformat() for t, _, _ in c["sens"]] and sv["scored"] == len(c["sens"])
+          and abs(sv["mae"]["last"] - sl) < 1e-9 and abs(sv["mae"]["mean5"] - s5) < 1e-9,
+          f"seed {s}: sensitivity dates, count and MAEs match")
+    P["last"].append(ml); P["mean5"].append(m5); S["last"].append(sl); S["mean5"].append(s5)
+
+agg = {k: sum(v) / len(v) for k, v in P.items()}
+verdict = "PASS" if agg["mean5"] <= 0.8 * agg["last"] else "FAIL"
+con = art["conclusion"]
+check(abs(con["primary"]["mae"]["last"] - agg["last"]) < 1e-9
+      and abs(con["primary"]["mae"]["mean5"] - agg["mean5"]) < 1e-9
+      and abs(con["primary"]["ratio"] - agg["mean5"] / agg["last"]) < 1e-9,
+      "conclusion aggregate (primary) matches recomputation")
+check(con["rule_verdict"] == verdict,
+      f"conclusion rule_verdict recomputed from frozen rule = {verdict}")
+check(isinstance(con.get("text"), str) and verdict.lower() in con["text"].lower(),
+      "conclusion text states the verdict")
+
+import report
+rows = series(3)
+base = report.score(rows, "primary")
+pert = copy.deepcopy(rows); pert[-1]["value"] = 99999
+check(report.score(pert, "primary")[:-1] == base[:-1], "past-only: perturbing last value leaves earlier errors unchanged (primary)")
+base = report.score(rows, "sensitivity")
+check(report.score(pert, "sensitivity")[:-1] == base[:-1], "past-only: same under sensitivity rule")
+
+with tempfile.TemporaryDirectory() as td:
+    outs = []
+    for n in (1, 2):
+        p = Path(td) / f"r{n}.json"
+        subprocess.run([sys.executable, str(Path(__file__).resolve().parent.parent / "report.py"), "--out", str(p)], check=True)
+        outs.append(p.read_bytes())
+    check(outs[0] == outs[1], "two fresh builds are byte-identical")
+    check(outs[0] == raw, "artifact on disk equals a fresh build byte for byte")
+
+print("RESULT:", "FAIL" if fails else "ALL CHECKS PASSED")
+sys.exit(1 if fails else 0)

+"""Build build/backtest.json: five-observation mean vs last observation, seeds 0-9.
+All errors are absolute errors in units per day. Only values before a target date are used.
+Rules: "primary"     = previous observed value / previous five observed values, every target
+                       from the sixth observation on is scored.
+       "sensitivity" = only targets whose five preceding calendar days are all observed."""
+import argparse
+from datetime import date, timedelta
+import json
+from pathlib import Path
+
+from run import series
+
+SEEDS = range(0, 10)
+THRESHOLD = 0.8  # frozen rule: PASS iff mean5 MAE <= 0.8 * last MAE (primary, aggregate)
+
+
+def score(rows, rule):
+    values = {date.fromisoformat(r["date"]): r["value"] for r in rows}
+    days = sorted(values)
+    out = []
+    for i, t in enumerate(days):
+        if rule == "primary":
+            if i < 5:
+                continue
+            prior = [values[d] for d in days[i - 5:i]]
+        elif rule == "sensitivity":
+            prev = [t - timedelta(days=k) for k in range(1, 6)]
+            if not all(p in values for p in prev):
+                continue
+            prior = [values[p] for p in reversed(prev)]
+        else:
+            raise ValueError(rule)
+        out.append({"date": t.isoformat(), "last": abs(values[t] - prior[-1]),
+                    "mean5": abs(values[t] - sum(prior) / 5)})
+    return out
+
+
+def mae(scored):
+    return {name: sum(e[name] for e in scored) / len(scored) for name in ("last", "mean5")}
+
+
+def run_for(seed):
+    rows = series(seed)
+    days = sorted(date.fromisoformat(r["date"]) for r in rows)
+    calendar_days = (days[-1] - days[0]).days + 1
+    prim, sens = score(rows, "primary"), score(rows, "sensitivity")
+    return {"seed": seed, "rows": len(rows), "calendar_days": calendar_days,
+            "gaps": calendar_days - len(rows),
+            "scored": len(prim), "scored_dates": [e["date"] for e in prim], "mae": mae(prim),
+            "sensitivity": {"scored": len(sens), "scored_dates": [e["date"] for e in sens],
+                            "mae": mae(sens)}}
+
+
+def aggregate(runs, key):
+    get = (lambda r: r["mae"]) if key == "primary" else (lambda r: r["sensitivity"]["mae"])
+    agg = {n: sum(get(r)[n] for r in runs) / len(runs) for n in ("last", "mean5")}
+    return {"mae": agg, "ratio": agg["mean5"] / agg["last"],
+            "seeds_where_mean5_better": sum(get(r)["mean5"] < get(r)["last"] for r in runs)}
+
+
+def build():
+    runs = [run_for(s) for s in SEEDS]
+    prim, sens = aggregate(runs, "primary"), aggregate(runs, "sensitivity")
+    verdict = "PASS" if prim["mae"]["mean5"] <= THRESHOLD * prim["mae"]["last"] else "FAIL"
+    reduction = 100 * (1 - prim["ratio"])
+    text = (f"Rule verdict {verdict}: the five-observation mean does not reduce MAE by at least 20% versus the "
+            f"last observation." if verdict == "FAIL" else
+            f"Rule verdict {verdict}: the five-observation mean reduces MAE by at least 20% versus the last observation.")
+    text += (f" Over seeds 0-9 (primary rule) last MAE = {prim['mae']['last']:.3f}, mean5 MAE = "
+             f"{prim['mae']['mean5']:.3f} units/day, ratio {prim['ratio']:.2f} ({reduction:+.0f}% change in MAE; "
+             f"negative = worse). Mean5 was better in {prim['seeds_where_mean5_better']} of {len(runs)} seeds. "
+             f"The history is not complete: every seed has {runs[0]['rows']} observations over "
+             f"{runs[0]['calendar_days']} calendar days ({runs[0]['gaps']} missing days). "
+             f"Sensitivity rule ratio {sens['ratio']:.2f}. Cause (inferred, not ablated): with a trend of "
+             f"1 unit/day the window mean lags by about 3 units, far above the +/-0.2 noise it removes.")
+    return {"method": {"errors": "absolute, units per day",
+                       "last": "previous observed value", "mean5": "mean of previous five observed values",
+                       "seeds": "0 through 9 inclusive",
+                       "frozen_rule": "PASS iff aggregate mean5 MAE <= 0.8 x last MAE (primary rule)"},
+            "runs": runs,
+            "conclusion": {"rule_verdict": verdict, "primary": prim, "sensitivity": sens, "text": text}}
+
+
+if __name__ == "__main__":
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--out", default="build/backtest.json")
+    out = Path(parser.parse_args().out)
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(build(), indent=2, sort_keys=True) + "\n", encoding="utf-8")
