+"""Build build/backtest.json: last vs mean5, seeds 0-9, units per day."""
+import json
+import sys
+from datetime import date
+from pathlib import Path
+
+from run import series
+
+SEEDS = range(10)
+THRESHOLD = 0.8  # frozen rule: mean5 MAE <= 0.8 * last MAE
+
+
+def score(rows, eligible):
+    out = []
+    for i in range(5, len(rows)):
+        if not eligible(rows, i):
+            continue
+        v = [r["value"] for r in rows]
+        out.append((rows[i]["date"], abs(v[i] - v[i - 1]), abs(v[i] - sum(v[i - 5:i]) / 5)))
+    return out
+
+
+def six_consecutive(rows, i):
+    d = [date.fromisoformat(r["date"]) for r in rows[i - 5:i + 1]]
+    return (d[-1] - d[0]).days == 5
+
+
+def summarize(scored):
+    n = len(scored)
+    last = sum(s[1] for s in scored) / n
+    mean5 = sum(s[2] for s in scored) / n
+    return {"scored": n, "first_date": scored[0][0], "last_date": scored[-1][0],
+            "mae_last": last, "mae_mean5": mean5, "ratio": mean5 / last}
+
+
+def build():
+    runs = []
+    for seed in SEEDS:
+        rows = series(seed)
+        ds = [date.fromisoformat(r["date"]) for r in rows]
+        gaps = sum((b - a).days != 1 for a, b in zip(ds, ds[1:]))
+        runs.append({
+            "seed": seed, "rows": len(rows), "calendar_days_spanned": (ds[-1] - ds[0]).days + 1,
+            "gaps": gaps,
+            "primary_six_consecutive_days": summarize(score(rows, six_consecutive)),
+            "secondary_all_observed_descriptive": summarize(score(rows, lambda r, i: True)),
+        })
+    ratios = [r["primary_six_consecutive_days"]["ratio"] for r in runs]
+    passes = [x <= THRESHOLD for x in ratios]
+    verdict = "PASS" if all(passes) else "FAIL"
+    conclusion = {
+        "rule": "mean5 MAE at most 0.8 times last-observation MAE, every seed 0-9, primary window",
+        "verdict": verdict, "seeds_passing": sum(passes), "seeds_total": len(passes),
+        "ratio_min": min(ratios), "ratio_max": max(ratios),
+        "summary": (f"{verdict}: mean5/last MAE ratio ranges {min(ratios):.2f} to {max(ratios):.2f} "
+                    f"across seeds 0-9; {sum(passes)} of {len(passes)} seeds meet the 0.8 threshold. "
+                    "The generated history has gaps (every 17th day is missing), so it does not have an "
+                    "observation for every calendar day. Smoothing did not reduce MAE; on a +1/day "
+                    "trend mean5 lags about 3 units."),
+    }
+    return {"units": "units per day", "runs": runs, "conclusion": conclusion}
+
+
+if __name__ == "__main__":
+    out = Path(sys.argv[1] if len(sys.argv) > 1 else "build/backtest.json")
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(build(), indent=2, sort_keys=True) + "\n")

+"""Independent check of build/backtest.json (re-derives everything from raw dates/values)."""
+import json
+import random
+import sys
+from datetime import date, timedelta
+
+path = sys.argv[1] if len(sys.argv) > 1 else "build/backtest.json"
+rep = json.load(open(path))
+assert sorted(rep) == ["conclusion", "runs", "units"], sorted(rep)
+assert [r["seed"] for r in rep["runs"]] == list(range(10))
+ratios = []
+for r in rep["runs"]:
+    rng = random.Random(r["seed"])  # regenerate independently of run.py
+    days, vals = [], []
+    for d in range(100):
+        v = 50 + d + rng.uniform(-0.2, 0.2)
+        if d and d % 17 == 0:
+            continue
+        days.append(d); vals.append(v)
+    gaps = sum(b - a != 1 for a, b in zip(days, days[1:]))
+    assert gaps > 0 and r["gaps"] == gaps and r["rows"] == len(days) == 95, (r["seed"], gaps)
+    assert r["calendar_days_spanned"] == 100
+    for key, ok in (("primary_six_consecutive_days", lambda i: days[i] - days[i - 5] == 5),
+                    ("secondary_all_observed_descriptive", lambda i: True)):
+        idx = [i for i in range(5, len(days)) if ok(i)]
+        el = sum(abs(vals[i] - vals[i - 1]) for i in idx) / len(idx)
+        em = sum(abs(vals[i] - sum(vals[i - 5:i]) / 5) for i in idx) / len(idx)
+        s = r[key]
+        assert s["scored"] == len(idx), (key, s["scored"], len(idx))
+        assert abs(s["mae_last"] - el) < 1e-9 and abs(s["mae_mean5"] - em) < 1e-9
+        assert abs(s["ratio"] - em / el) < 1e-9
+        d0 = date(2025, 1, 1)
+        assert s["first_date"] == (d0 + timedelta(days=days[idx[0]])).isoformat()
+        assert s["last_date"] == (d0 + timedelta(days=days[idx[-1]])).isoformat()
+        if key.startswith("primary"):
+            assert len(idx) == 65  # A2
+            ratios.append(em / el)
+verdict = "PASS" if all(x <= 0.8 for x in ratios) else "FAIL"
+c = rep["conclusion"]
+assert c["verdict"] == verdict and c["seeds_passing"] == sum(x <= 0.8 for x in ratios)
+assert c["summary"].startswith(verdict)
+print("OK", verdict, "ratio range", round(min(ratios), 3), round(max(ratios), 3))
