+def build_export(rows):
+    """Group rows by invoice id (source line order kept), invoices sorted by id.
+
+    Amounts are per-line rounded (ledger rule) and serialised as strings.
+    """
+    grouped = {}
+    for row in rows:
+        grouped.setdefault(row["invoice"], []).append(row)
+    invoices = []
+    for invoice_id in sorted(grouped):
+        lines = []
+        for row in grouped[invoice_id]:
+            line = {"item": row["item"], "amount": str(line_total(row))}
+            if row.get("tax_class"):
+                line["tax_class"] = row["tax_class"]
+            lines.append(line)
+        invoices.append({"id": invoice_id, "total": str(total(grouped[invoice_id])), "lines": lines})
+    return {"invoices": invoices}
+

+def write_export(rows, path):
+    path = Path(path)
+    path.parent.mkdir(parents=True, exist_ok=True)
+    path.write_text(json.dumps(build_export(rows), indent=2) + "\n", encoding="utf-8")
+    return path
