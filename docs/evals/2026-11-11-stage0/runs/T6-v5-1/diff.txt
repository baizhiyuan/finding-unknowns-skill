+"""Reusable invoice export builder. Ledger rule: round each line, then sum (see docs/accounting.md)."""
+import argparse
+import json
+from itertools import groupby
+from pathlib import Path
+
+from invoice import line_total, load, total
+
+
+def build_export(rows):
+    ordered = sorted(enumerate(rows), key=lambda p: (p[1]["invoice"], p[0]))  # stable: preserves source order
+    invoices = []
+    for inv_id, group in groupby((r for _, r in ordered), key=lambda r: r["invoice"]):
+        group = list(group)
+        lines = []
+        for r in group:
+            line = {"item": r["item"], "amount": str(line_total(r))}
+            if r.get("tax_class"):
+                line["tax_class"] = r["tax_class"]
+            lines.append(line)
+        invoices.append({"id": inv_id, "total": str(total(group)), "lines": lines})
+    return {"invoices": invoices}
+
+
+def write_export(source, destination):
+    payload = build_export(load(source))
+    destination = Path(destination)
+    destination.parent.mkdir(parents=True, exist_ok=True)
+    destination.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8")
+    return payload
+
+
+if __name__ == "__main__":
+    ap = argparse.ArgumentParser(description="Write the invoice batch export.")
+    ap.add_argument("--source", default="data/lines.csv")
+    ap.add_argument("--output", default="build/invoices.json")
+    args = ap.parse_args()
+    write_export(args.source, args.output)
+    print(f"wrote {args.output}")

+import json
+import subprocess
+import sys
+from decimal import Decimal
+
+from export import build_export, write_export
+from invoice import load, total
+
+
+def row(inv, item, qty, price, tax="standard"):
+    return {"invoice": inv, "item": item, "quantity": qty, "unit_price": price, "tax_class": tax}
+
+
+def test_batch_matches_ledger_rule_literals():
+    out = build_export(load("data/lines.csv"))
+    assert out == {"invoices": [{
+        "id": "I100",
+        "total": "0.92",
+        "lines": [
+            {"item": "Usage A", "amount": "0.34", "tax_class": "standard"},
+            {"item": "Usage B", "amount": "0.34", "tax_class": "standard"},
+            {"item": "Usage C", "amount": "0.34", "tax_class": "standard"},
+            {"item": "Refund", "amount": "-0.10", "tax_class": "exempt"},
+        ],
+    }]}
+
+
+def test_lines_sum_to_total():
+    for inv in build_export(load("data/lines.csv"))["invoices"]:
+        assert sum(Decimal(l["amount"]) for l in inv["lines"]) == Decimal(inv["total"])
+
+
+def test_sorted_ids_and_source_order():
+    rows = [row("B2", "z", "1", "1.00"), row("A1", "y", "1", "2.00"), row("B2", "a", "2", "0.505")]
+    out = build_export(rows)["invoices"]
+    assert [i["id"] for i in out] == ["A1", "B2"]
+    assert [l["item"] for l in out[1]["lines"]] == ["z", "a"]
+    assert out[1]["lines"][1]["amount"] == "1.01"
+    assert out[1]["total"] == "2.01"
+
+
+def test_total_api_unchanged():
+    assert total([{"quantity": "1", "unit_price": "0.335"}] * 3) == Decimal("1.02")
+
+
+def test_command_writes_artifact(tmp_path):
+    dest = tmp_path / "sub" / "invoices.json"
+    write_export("data/lines.csv", dest)
+    assert json.loads(dest.read_text()) == build_export(load("data/lines.csv"))
+    r = subprocess.run([sys.executable, "export.py", "--output", str(tmp_path / "cli.json")],
+                       capture_output=True, text=True)
+    assert r.returncode == 0
+    assert json.loads((tmp_path / "cli.json").read_text())["invoices"][0]["total"] == "0.92"
