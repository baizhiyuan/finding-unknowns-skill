+def build_export(rows):
+    """Search export: every (isbn, edition) row, inactive ones flagged via "active".
+
+    Sorted by (displayed title, isbn, edition) in Python string order, so output
+    does not depend on import order.
+    """
+    books = [
+        {"isbn": r["isbn"], "edition": r["edition"], "title": r["title"],
+         "active": r["active"], "key": search_key(r["title"])}
+        for r in rows
+    ]
+    books.sort(key=lambda b: (b["title"], b["isbn"], b["edition"]))
+    return {"books": books}
