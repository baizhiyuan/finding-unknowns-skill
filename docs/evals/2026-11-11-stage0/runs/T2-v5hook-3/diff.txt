+"""Locked contract checks for the prefix-search export. Not edited by the implementer."""
+import json
+from pathlib import Path
+
+from catalog import load, search
+
+EXPORT = Path("build/search.json")
+FIELDS = {"isbn", "edition", "title", "active", "key"}
+
+
+def export():
+    return json.loads(EXPORT.read_text(encoding="utf-8"))["books"]
+
+
+def test_rows_fields_and_keys():
+    src = load()
+    out = export()
+    assert {(r["isbn"], r["edition"]) for r in out} == {(r["isbn"], r["edition"]) for r in src}
+    assert len(out) == len(src) == 4
+    by_id = {(r["isbn"], r["edition"]): r for r in src}
+    for r in out:
+        assert set(r) == FIELDS
+        s = by_id[(r["isbn"], r["edition"])]
+        assert r["title"] == s["title"] and r["active"] == s["active"]
+        assert r["key"] == s["title"].casefold()
+
+
+def test_order():
+    src = load()
+    want = [(r["isbn"], r["edition"]) for r in sorted(src, key=lambda r: (r["title"], r["isbn"], r["edition"]))]
+    assert [(r["isbn"], r["edition"]) for r in export()] == want
+
+
+def test_prefix_search_uses_key_not_title():
+    rows = export()
+    for q in ("strasse", "STRAS", "straß"):
+        got = search(q, rows)
+        assert [(r["isbn"], r["edition"]) for r in got] == [
+            ("9780000000011", "large-print"), ("9780000000011", "paper")], q
+        assert all(r["title"] == "Straße Atlas" for r in got)
+    assert [r["title"] for r in search("app", rows)] == ["apple Almanac"]
+    assert [r["title"] for r in search("zeb", rows)] == ["Zebra Tales"]  # inactive stock is searchable
+    assert search("zzz", rows) == []
