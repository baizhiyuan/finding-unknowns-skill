+"""Build the replacement table preview. Never touches the live table."""
+import csv
+from datetime import datetime
+from pathlib import Path
+
+from attendance import as_utc, load
+
+FIELDS = ["badge", "session", "start_utc"]
+
+
+def normalize(rows):
+    """One row per (badge, session): latest observed_at instant wins; if that
+    observation is cancelled the pair is dropped. Ties keep the later input row."""
+    latest = {}
+    for r in rows:
+        key = (r["badge"], r["session"])
+        stamp = datetime.fromisoformat(r["observed_at"])
+        if key not in latest or stamp >= latest[key][0]:
+            latest[key] = (stamp, r)
+    return [{"badge": b, "session": s, "start_utc": as_utc(r["started_at"])}
+            for (b, s), (_, r) in sorted(latest.items()) if r["status"] != "cancelled"]
+
+
+def write_preview(rows, path="build/normalized.csv"):
+    path = Path(path)
+    path.parent.mkdir(parents=True, exist_ok=True)
+    with path.open("w", newline="") as stream:
+        writer = csv.DictWriter(stream, fieldnames=FIELDS, lineterminator="\n")
+        writer.writeheader()
+        writer.writerows(rows)
+
+
+if __name__ == "__main__":
+    import sys
+    if len(sys.argv) != 2:
+        raise SystemExit("usage: python normalize.py LEGACY.csv  (writes build/normalized.csv)")
+    write_preview(normalize(load(sys.argv[1])))

+import csv
+from datetime import datetime
+from normalize import normalize, write_preview
+
+
+def row(legacy_id, badge, session, started, observed, status="active"):
+    return {"legacy_id": legacy_id, "badge": badge, "session": session,
+            "started_at": started, "observed_at": observed, "status": status}
+
+
+def test_shared_legacy_id_does_not_merge_different_pairs():
+    rows = [row("1", "A", "am", "2025-06-01T08:00:00Z", "2025-06-01T08:01:00Z"),
+            row("1", "B", "am", "2025-06-01T08:00:00Z", "2025-06-01T08:01:00Z")]
+    assert [r["badge"] for r in normalize(rows)] == ["A", "B"]
+
+
+def test_duplicate_pair_keeps_latest_observed_instant_across_offsets():
+    rows = [row("1", "A", "am", "2025-06-01T08:00:00Z", "2025-06-01T10:00:00+02:00"),
+            row("2", "A", "am", "2025-06-01T08:05:00Z", "2025-06-01T09:00:00Z")]
+    out = normalize(rows)
+    assert out == [{"badge": "A", "session": "am", "start_utc": "2025-06-01T08:05:00Z"}]
+
+
+def test_cancelled_latest_observation_is_dropped():
+    rows = [row("1", "A", "am", "2025-06-01T08:00:00Z", "2025-06-01T08:01:00Z", "cancelled")]
+    assert normalize(rows) == []
+
+
+def test_output_sorted_and_utc():
+    rows = [row("1", "B", "am", "2025-06-01T00:15:00+02:00", "2025-06-01T08:00:00Z"),
+            row("2", "A", "pm", "2025-06-01T14:00:00Z", "2025-06-01T14:01:00Z")]
+    out = normalize(rows)
+    assert [(r["badge"], r["session"]) for r in out] == [("A", "pm"), ("B", "am")]
+    assert out[1]["start_utc"] == "2025-05-31T22:15:00Z"
+
+
+def test_preview_file_format(tmp_path):
+    path = tmp_path / "build" / "n.csv"
+    write_preview(normalize([row('1', 'B', 'am', '2025-06-01T00:15:00+02:00', '2025-06-01T08:00:00Z')]), path)
+    lines = path.read_text().splitlines()
+    assert lines[0] == "badge,session,start_utc"
+    assert all(l.endswith("Z") for l in lines[1:])
