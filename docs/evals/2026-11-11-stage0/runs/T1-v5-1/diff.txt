+def select_cheapest(parcel):
+    """Cheapest quote for a parcel; ties go to the first service in table order."""
+    options = quotes(parcel)
+    if not options:
+        return {"id": parcel["id"], "status": "unavailable",
+                "reason": f"no service covers zone '{parcel['zone']}'"}
+    best = min(options, key=lambda q: q["cost"])  # min keeps the first of equal costs
+    return {"id": parcel["id"], "status": "selected", "service": best["service"],
+            "days": best["days"], "cost": best["cost"]}
+
+
+def select_batch(parcels):
+    return {"parcels": [select_cheapest(p) for p in parcels]}
