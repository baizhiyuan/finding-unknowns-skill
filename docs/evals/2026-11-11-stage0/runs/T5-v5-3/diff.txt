+import csv
+from datetime import datetime
+from pathlib import Path
+
+from attendance import as_utc, load
+
+
+def normalize(rows):
+    latest = {}
+    for row in rows:
+        key = (row["badge"], row["session"])
+        seen = latest.get(key)
+        stamp = datetime.fromisoformat(row["observed_at"])
+        if seen is None or stamp > seen[0]:
+            latest[key] = (stamp, row)
+        elif stamp == seen[0]:
+            raise ValueError(f"tie on observed_at for {key}")
+    return [{"badge": b, "session": s, "start_utc": as_utc(row["started_at"])}
+            for (b, s), (_, row) in sorted(latest.items()) if row["status"] != "cancelled"]
+
+
+def write(rows, target="build/normalized.csv"):
+    target = Path(target)
+    target.parent.mkdir(parents=True, exist_ok=True)
+    with target.open("w", newline="") as stream:
+        writer = csv.DictWriter(stream, ["badge", "session", "start_utc"], lineterminator="\n")
+        writer.writeheader()
+        writer.writerows(rows)
+
+
+if __name__ == "__main__":
+    write(normalize(load()))
