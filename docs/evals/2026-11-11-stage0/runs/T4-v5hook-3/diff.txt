+Recomputes every MAE from listed scored dates with calendar-date lookups (not row arithmetic)."""
+import hashlib, json, sys
+from datetime import date, timedelta
+from pathlib import Path
+sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

+
+raw = Path(sys.argv[1]).read_bytes()
+rep = json.loads(raw)

+assert [r["seed"] for r in rep["runs"]] == list(range(10))
+pooled = {"last": 0.0, "mean5": 0.0}
+n = 0
+reductions = []
+for r in rep["runs"]:
+    val = {date.fromisoformat(x["date"]): x["value"] for x in series(r["seed"])}
+    span = (max(val) - min(val)).days + 1
+    assert r["rows"] == len(val) and r["gaps"] == span - len(val), "coverage not measured from dates"
+    assert r["scored"] == len(r["scored_dates"])
+    l = m = 0.0
+    for s in r["scored_dates"]:
+        t = date.fromisoformat(s)
+        prev = [t - timedelta(days=k) for k in range(1, 6)]
+        assert t in val and all(p in val for p in prev), f"ineligible date scored: {s}"
+        l += abs(val[t] - val[prev[0]])
+        m += abs(val[t] - sum(val[p] for p in prev) / 5)
+    k = r["scored"]
+    assert abs(l / k - r["mae"]["last"]) < 1e-9 and abs(m / k - r["mae"]["mean5"]) < 1e-9, r["seed"]
+    pooled["last"] += l; pooled["mean5"] += m; n += k
+    reductions.append(1 - (m / k) / (l / k))
+pl, pm = pooled["last"] / n, pooled["mean5"] / n
+pooled_red = 1 - pm / pl
+supported = pooled_red >= 0.20 and all(x >= 0.20 for x in reductions)
+head = rep["conclusion"].split(":")[0].strip().upper()
+assert head == ("SUPPORTED" if supported else "NOT SUPPORTED"), (head, supported)

+"""Backtest report: 5-observation mean vs last observation on calendar-eligible targets.
+
+A target is eligible only if it and its five preceding observations occupy six
+consecutive calendar days. Missing days are never filled. All errors are in units/day.
+"""
+import argparse
+from datetime import date
+import json
+from pathlib import Path
+
+from run import series
+
+THRESHOLD = 0.20  # frozen decision rule: mean5 MAE at least 20% below last MAE
+
+
+def eligible_indices(rows):
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    return [i for i in range(5, len(rows)) if (days[i] - days[i - 5]).days == 5]
+
+
+def score_rows(rows, indices=None):
+    values = [r["value"] for r in rows]
+    indices = eligible_indices(rows) if indices is None else indices
+    return [{"date": rows[i]["date"],
+             "last": abs(values[i] - values[i - 1]),
+             "mean5": abs(values[i] - sum(values[i - 5:i]) / 5)} for i in indices]
+
+
+def _mae(scored):
+    return {name: sum(e[name] for e in scored) / len(scored) for name in ("last", "mean5")}
+
+
+def _reduction(mae):
+    return 1 - mae["mean5"] / mae["last"]
+
+
+def _run(seed):
+    rows = series(seed)
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    scored = score_rows(rows)
+    return {"seed": seed, "rows": len(rows),
+            "gaps": (days[-1] - days[0]).days + 1 - len(rows),  # measured from dates
+            "scored": len(scored), "scored_dates": [e["date"] for e in scored],
+            "mae": _mae(scored)}, scored
+
+
+def build_report(seeds):
+    runs, pooled = [], []
+    for seed in seeds:
+        run, scored = _run(seed)
+        runs.append(run)
+        pooled += scored
+    pm = _mae(pooled)
+    reds = [_reduction(r["mae"]) for r in runs]
+    supported = _reduction(pm) >= THRESHOLD and all(x >= THRESHOLD for x in reds)
+    verdict = "SUPPORTED" if supported else "NOT SUPPORTED"
+    conclusion = (
+        f"{verdict}: the five-observation mean {'cuts' if _reduction(pm) > 0 else 'does not cut'} "
+        f"MAE by at least 20% versus the last observation. Pooled over {len(runs)} seeds and "
+        f"{len(pooled)} eligible dates, MAE (units/day) is {pm['last']:.3f} for last and "
+        f"{pm['mean5']:.3f} for mean5, a change of {_reduction(pm):+.1%} "
+        f"(per-seed {min(reds):+.1%} to {max(reds):+.1%}; negative means mean5 is worse). "
+        f"Targets are eligible only when the target and its five preceding observations occupy "
+        f"six consecutive calendar days; each seed has {runs[0]['gaps']} missing calendar days "
+        f"and no days are filled. On a +1 unit/day trend a trailing mean lags the level, so "
+        f"smoothing adds error here.")
+    return {"runs": runs, "conclusion": conclusion}
+
+
+def sensitivity(seeds):
+    """All-rows variant (windows may span gaps), for comparison only; never delivered."""
+    out = []
+    for seed in seeds:
+        rows = series(seed)
+        out.append({"seed": seed, "scored": len(rows) - 5,
+                    "mae": _mae(score_rows(rows, list(range(5, len(rows)))))})
+    return out
+
+
+if __name__ == "__main__":
+    parser = argparse.ArgumentParser()
+    parser.add_argument("--out", default="build/backtest.json")
+    parser.add_argument("--sensitivity-out", default="build/sensitivity.json")
+    args = parser.parse_args()
+    seeds = range(10)
+    for path, payload in ((args.out, build_report(seeds)), (args.sensitivity_out, sensitivity(seeds))):
+        Path(path).parent.mkdir(parents=True, exist_ok=True)
+        Path(path).write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8")

+"""Contract checks for report.py. Written before the implementation; not edited by it."""
+from datetime import date, timedelta
+from run import series
+from report import eligible_indices, score_rows, build_report
+
+
+def mk(days, start=date(2025, 1, 1), value=lambda d: float(d)):
+    return [{"date": (start + timedelta(days=d)).isoformat(), "value": value(d)} for d in days]
+
+
+def calendar_eligible(rows):
+    """Independent implementation: calendar-set membership, no row arithmetic."""
+    have = {date.fromisoformat(r["date"]) for r in rows}
+    return [r["date"] for r in rows
+            if all(date.fromisoformat(r["date"]) - timedelta(days=k) in have for k in range(0, 6))]
+
+
+def test_rule_on_handbuilt_gap():
+    rows = mk([0, 1, 2, 3, 4, 5, 6, 8, 9, 10, 11, 12, 13, 14])  # day 7 missing
+    scored = [rows[i]["date"] for i in eligible_indices(rows)]
+    assert scored == [mk([d])[0]["date"] for d in (5, 6, 13, 14)]
+    assert scored == [d for d in calendar_eligible(rows) if d > rows[4]["date"]]
+
+
+def test_first_five_never_scored_and_no_gap_scores_all_after():
+    rows = mk(range(10))
+    assert eligible_indices(rows) == [5, 6, 7, 8, 9]
+
+
+def test_65_eligible_per_seed_and_matches_independent_count():
+    for seed in range(10):
+        rows = series(seed)
+        assert len(rows) == 95
+        idx = eligible_indices(rows)
+        assert [rows[i]["date"] for i in idx] == [d for d in calendar_eligible(rows)
+                                                  if d >= rows[5]["date"]]
+        assert len(idx) == 65
+
+
+def test_noiseless_trend_errors_and_same_dates():
+    rows = mk(range(12))
+    out = score_rows(rows)
+    assert [e["date"] for e in out] == [rows[i]["date"] for i in eligible_indices(rows)]
+    assert all(e["last"] == 1.0 and e["mean5"] == 3.0 for e in out)
+
+
+def test_only_past_values():
+    rows = mk(range(12))
+    before = score_rows(rows)[0]
+    rows[5]["value"] = 99999.0           # target itself
+    rows[6]["value"] = -5.0              # later values
+    after = score_rows(rows)[0]
+    # forecasts unchanged: error moves only through the target's own value
+    forecast_last = rows[4]["value"]
+    assert after["last"] == abs(99999.0 - forecast_last)
+    assert after["mean5"] == abs(99999.0 - sum(r["value"] for r in rows[0:5]) / 5)
+    assert before["date"] == after["date"]
+
+

+    assert set(rep) == {"runs", "conclusion"}
+    assert [r["seed"] for r in rep["runs"]] == list(range(10))
+    for r in rep["runs"]:
+        assert r["rows"] == 95 and r["gaps"] == 5 and r["scored"] == 65
+        assert len(r["scored_dates"]) == 65
+        assert set(r["mae"]) == {"last", "mean5"}
+    assert isinstance(rep["conclusion"], str) and rep["conclusion"]
