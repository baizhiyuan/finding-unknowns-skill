+def eligible_errors(rows, window=20):
+    """Errors for the final `window` targets whose 5 predecessors are consecutive calendar days."""
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    ok = {r["date"] for i, r in enumerate(rows)
+          if i >= 5 and (days[i] - days[i-5]).days == 5}
+    eligible = [e for e in errors(rows) if e["date"] in ok]
+    return eligible, eligible[-window:]

+def run_report(seed, window=20):
+    rows = series(seed)
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    span = (days[-1] - days[0]).days + 1
+    eligible, scored = eligible_errors(rows, window)
+    mae = {n: sum(e[n] for e in scored) / len(scored) for n in ("last", "mean5")}
+    return {"seed": seed, "rows": len(rows), "calendar_days": span,
+            "missing_days": span - len(rows),
+            "eligible_targets": len(eligible), "scored_count": len(scored),
+            "scored_dates": [e["date"] for e in scored],
+            "mae_last": mae["last"], "mae_mean5": mae["mean5"],
+            "ratio_mean5_over_last": mae["mean5"] / mae["last"]}
+

+def build_report(seeds=range(10), window=20):
+    runs = [run_report(s, window) for s in seeds]
+    ratios = [r["ratio_mean5_over_last"] for r in runs]
+    worst_for_claim = min(ratios)
+    holds = all(x <= 0.8 for x in ratios)
+    pooled_last = sum(r["mae_last"] for r in runs) / len(runs)
+    pooled_mean5 = sum(r["mae_mean5"] for r in runs) / len(runs)
+    return {"units": "units per day", "window": f"final {window} eligible targets",
+            "eligibility": "target and 5 preceding observations span 6 consecutive calendar days; no gap filling",
+            "runs": runs,
+            "pooled": {"mae_last": pooled_last, "mae_mean5": pooled_mean5,
+                       "ratio_mean5_over_last": pooled_mean5 / pooled_last,
+                       "min_run_ratio": worst_for_claim, "max_run_ratio": max(ratios)},
+            "conclusion": (
+                ("Hypothesis supported: mean5 MAE <= 0.8x last in every seed." if holds else
+                 "Hypothesis NOT supported: the 20% MAE reduction (ratio <= 0.8) does not hold; "
+                 f"mean5/last ratio ranged {min(ratios):.2f} to {max(ratios):.2f} across seeds "
+                 f"(pooled {pooled_mean5/pooled_last:.2f}). On an upward trend the 5-point mean lags, "
+                 "so it is worse than the last observation. "
+                 f"Calendar coverage is incomplete: {runs[0]['missing_days']} missing days per seed."))}
