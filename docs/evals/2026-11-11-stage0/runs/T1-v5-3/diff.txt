+def select(parcel):
+    """Cheapest quote; equal cost goes to the alphabetically first service name."""
+    options = quotes(parcel)
+    if not options:
+        return {"id": parcel["id"], "status": "unavailable",
+                "reason": f"no service covers zone '{parcel['zone']}'"}
+    best = min(options, key=lambda q: (q["cost"], q["service"]))
+    return {"id": parcel["id"], "status": "selected", **best}
+
+
+def select_all(parcels):
+    return {"parcels": [select(p) for p in parcels]}
+
