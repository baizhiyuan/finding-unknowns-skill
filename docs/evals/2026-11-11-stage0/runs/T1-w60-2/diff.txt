+def select(parcel):
+    """Cheapest quote for one parcel as an output entry.
+
+    Equal prices: the service listed first in SERVICES wins (no business
+    preference is set). No qualifying service gives status "unavailable".
+    """
+    options = quotes(parcel)
+    if not options:
+        return {"id": parcel["id"], "status": "unavailable",
+                "reason": f"no service covers zone {parcel['zone']!r}"}
+    best = min(options, key=lambda q: q["cost"])
+    return {"id": parcel["id"], "status": "selected", "service": best["service"],
+            "days": best["days"], "cost": best["cost"]}
+
+
+def select_all(parcels):
+    return {"parcels": [select(p) for p in parcels]}
