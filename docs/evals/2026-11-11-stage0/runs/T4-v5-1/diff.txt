+"""Build the mean5-vs-last backtest report (seeds 0-9). Usage: python report.py build/backtest.json"""
+from datetime import date, timedelta
+import json
+import sys
+from pathlib import Path
+
+from run import errors, series
+
+SEEDS = range(10)
+
+
+def mean(xs):
+    return sum(xs) / len(xs)
+
+
+def calendar_gaps(rows):
+    """Missing calendar dates between first and last generated date (measured from the data)."""
+    have = {date.fromisoformat(r["date"]) for r in rows}
+    d, end, missing = min(have), max(have), []
+    while d <= end:
+        if d not in have:
+            missing.append(d.isoformat())
+        d += timedelta(days=1)
+    return missing
+
+
+def gap_free_errors(rows):
+    """Score a date only if the 5 preceding calendar days are all observed."""
+    by_date = {date.fromisoformat(r["date"]): r["value"] for r in rows}
+    out = []
+    for d in sorted(by_date):
+        prev = [d - timedelta(days=k) for k in range(1, 6)]
+        if all(p in by_date for p in prev):
+            est_last = by_date[prev[0]]
+            est_mean5 = mean([by_date[p] for p in prev])
+            out.append({"date": d.isoformat(), "last": abs(by_date[d] - est_last),
+                        "mean5": abs(by_date[d] - est_mean5)})
+    return out
+
+
+def summarize(errs):
+    last, m5 = mean([e["last"] for e in errs]), mean([e["mean5"] for e in errs])
+    return {"scored": len(errs), "scored_dates": [e["date"] for e in errs],
+            "mae_last": last, "mae_mean5": m5,
+            "mean5_over_last": m5 / last, "mae_reduction_pct": 100 * (last - m5) / last}
+
+
+def build():
+    runs = []
+    for seed in SEEDS:
+        rows = series(seed)
+        missing = calendar_gaps(rows)
+        runs.append({"seed": seed, "rows": len(rows), "calendar_days_observed_pct":
+                     100 * len(rows) / (len(rows) + len(missing)),
+                     "missing_dates": missing, "gaps": len(missing),
+                     "primary_gap_free_calendar": summarize(gap_free_errors(rows)),
+                     "secondary_observed_order": summarize(errors(rows))})
+    p = [r["primary_gap_free_calendar"]["mae_reduction_pct"] for r in runs]
+    s = [r["secondary_observed_order"]["mae_reduction_pct"] for r in runs]
+    pl = mean([r["primary_gap_free_calendar"]["mae_last"] for r in runs])
+    pm = mean([r["primary_gap_free_calendar"]["mae_mean5"] for r in runs])
+    conclusion = (
+        f"Claim NOT supported. Under the verdict rule (gap-free calendar), mean5 MAE averages {pm:.3f} vs "
+        f"{pl:.3f} for last observation across seeds 0-9: mean5 is worse by {-mean(p):.1f}% "
+        f"(per-seed change {min(p):.1f}% to {max(p):.1f}%), not the predicted >=20% reduction. "
+        f"Observed-order rule (baseline) agrees: change {min(s):.1f}% to {max(s):.1f}%. "
+        f"The history is not complete: every seed has {runs[0]['gaps']} missing calendar days "
+        f"({', '.join(runs[0]['missing_dates'])}). Cause: the +1/day trend makes the 5-day mean lag "
+        f"~3 units, far above the +-0.2 noise it would smooth. Applies to this synthetic trend series only.")
+    return {"units": "units per day", "metric": "MAE", "seeds": list(SEEDS),
+            "verdict_rule": "primary_gap_free_calendar", "runs": runs, "conclusion": conclusion}
+
+
+if __name__ == "__main__":
+    out = Path(sys.argv[1])
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(build(), indent=2, sort_keys=True) + "\n")
