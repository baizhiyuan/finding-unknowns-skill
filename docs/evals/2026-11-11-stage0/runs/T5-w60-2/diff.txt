+import csv
+from datetime import datetime
+from pathlib import Path
+import sys
+
+from attendance import as_utc, load
+
+
+def normalize(rows):
+    """Latest observed_at (as an instant) per (badge, session); drop pair if that one is cancelled."""
+    latest = {}
+    for r in rows:
+        key = (r["badge"], r["session"])
+        when = datetime.fromisoformat(r["observed_at"])
+        if key not in latest or when > latest[key][0]:
+            latest[key] = (when, r)
+    return [{"badge": b, "session": s, "start_utc": as_utc(r["started_at"])}
+            for (b, s), (_, r) in sorted(latest.items()) if r["status"] != "cancelled"]
+
+
+def write(rows, out):
+    out = Path(out)
+    out.parent.mkdir(parents=True, exist_ok=True)
+    with out.open("w", newline="") as stream:
+        w = csv.DictWriter(stream, fieldnames=["badge", "session", "start_utc"], lineterminator="\n")
+        w.writeheader()
+        w.writerows(rows)
+
+
+if __name__ == "__main__":
+    write(normalize(load(sys.argv[1] if len(sys.argv) > 1 else "data/attendance.csv")), "build/normalized.csv")
