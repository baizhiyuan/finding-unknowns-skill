+"""Paired backtest: last vs mean5 on eligible targets (6 consecutive calendar days). Units/day."""
+import json
+import sys
+from datetime import date
+from pathlib import Path
+
+sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
+from run import experiment, series  # noqa: E402
+
+SEEDS = range(10)
+
+
+def run_seed(seed):
+    rows = series(seed)
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    vals = [r["value"] for r in rows]
+    gaps = sum((b - a).days != 1 for a, b in zip(days, days[1:]))
+    scored = []
+    for i in range(5, len(rows)):
+        if (days[i] - days[i - 5]).days == 5:  # 6 observations spanning 5 days, all distinct => consecutive
+            scored.append((days[i].isoformat(), abs(vals[i] - vals[i - 1]),
+                           abs(vals[i] - sum(vals[i - 5:i]) / 5)))
+    n = len(scored)
+    mae_last = sum(s[1] for s in scored) / n
+    mae_mean5 = sum(s[2] for s in scored) / n
+    return {"seed": seed, "rows": len(rows), "calendar_gaps": gaps, "scored": n,
+            "excluded_targets": len(rows) - 5 - n,
+            "scored_dates": [s[0] for s in scored],
+            "mae_last": mae_last, "mae_mean5": mae_mean5,
+            "ratio_mean5_over_last": mae_mean5 / mae_last}
+
+
+def main(out):
+    runs = [run_seed(s) for s in SEEDS]
+    ml = sum(r["mae_last"] for r in runs) / len(runs)
+    mm = sum(r["mae_mean5"] for r in runs) / len(runs)
+    ratio = mm / ml
+    rule = "PASS" if mm <= 0.8 * ml else "FAIL"
+    concl = (f"Rule (mean5 MAE <= 0.8 x last MAE, mean over seeds 0-9): {rule}. "
+             f"Mean MAE last={ml:.4f}, mean5={mm:.4f} units/day; ratio={ratio:.3f} "
+             f"(mean5 is {100*(ratio-1):+.1f}% vs last). Eligible = target and 5 preceding observations "
+             f"on 6 consecutive calendar days; no gap filling.")
+    sens = [experiment(s)["mae"] for s in SEEDS]
+    sensitivity = {"rule": "previous 5 observed values regardless of calendar gaps; all targets from the 6th observation (90 per seed); not the headline",
+                   "mean_mae_last": sum(m["last"] for m in sens) / 10,
+                   "mean_mae_mean5": sum(m["mean5"] for m in sens) / 10}
+    report = {"runs": runs, "conclusion": concl, "sensitivity": sensitivity,
+              "summary": {"mean_mae_last": ml, "mean_mae_mean5": mm, "ratio": ratio, "rule_verdict": rule}}
+    Path(out).parent.mkdir(parents=True, exist_ok=True)
+    Path(out).write_text(json.dumps(report, indent=2, sort_keys=True) + "\n")
+    print(concl)
+
+
+if __name__ == "__main__":
+    main(sys.argv[1])
