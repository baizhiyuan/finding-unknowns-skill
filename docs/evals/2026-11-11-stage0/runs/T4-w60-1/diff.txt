+def eligible_errors(rows, window=20):
+    """Errors for the final `window` targets whose 5 predecessors and the target
+    are six consecutive calendar days. Missing days are never filled."""
+    days = [date.fromisoformat(r["date"]) for r in rows]
+    values = [r["value"] for r in rows]
+    out = []
+    for i in range(5, len(rows)):
+        if (days[i] - days[i-5]).days != 5:  # dates strictly increase, so 5 days apart means 6 consecutive
+            continue
+        out.append({"date": rows[i]["date"], "last": abs(values[i] - values[i-1]),
+                    "mean5": abs(values[i] - sum(values[i-5:i]) / 5)})
+    return out[-window:]
+
+
+def window_report(seed, window=20):
+    rows = series(seed)
+    scored = eligible_errors(rows, window)
+    all_days = {date.fromisoformat(r["date"]) for r in rows}
+    span = (max(all_days) - min(all_days)).days + 1
+    mae = {n: sum(e[n] for e in scored) / len(scored) for n in ("last", "mean5")}
+    return {"seed": seed, "rows": len(rows), "calendar_days": span,
+            "missing_days": span - len(rows),
+            "scored": len(scored), "first_scored": scored[0]["date"],
+            "last_scored": scored[-1]["date"], "mae": mae,
+            "ratio_mean5_to_last": mae["mean5"] / mae["last"]}
+
+
+def build_report(seeds=range(10), threshold=0.8):
+    runs = [window_report(s) for s in seeds]
+    mean = {n: sum(r["mae"][n] for r in runs) / len(runs) for n in ("last", "mean5")}
+    ratio = mean["mean5"] / mean["last"]
+    wins = sum(r["ratio_mean5_to_last"] <= threshold for r in runs)
+    return {"units": "units per day", "metric": "MAE",
+            "rules": "final 20 eligible targets; eligible = target + 5 predecessors on 6 consecutive calendar days; no gap filling",
+            "runs": runs, "mean_mae": mean, "ratio_mean5_to_last": ratio,
+            "seeds_meeting_threshold": wins,
+            "conclusion": (f"Hypothesis (mean5 MAE <= {threshold} x last MAE) "
+                           + ("supported" if wins == len(runs) and ratio <= threshold else "NOT supported")
+                           + f": mean5/last = {ratio:.2f} on average; {wins}/{len(runs)} seeds meet the threshold. "
+                           "The input history has missing calendar days (5 per seed), so scoring excludes ineligible targets.")}
