+"""Locked contract checks for the prefix-search export (literal expectations, not derived from catalog.py)."""
+import json
+import subprocess
+import sys
+from pathlib import Path
+
+from catalog import build_export, search
+
+ROOT = Path(__file__).resolve().parent.parent
+
+EXPECTED = [
+    {"isbn": "9780000000011", "edition": "large-print", "title": "Straße Atlas", "active": True, "key": "strasse atlas"},
+    {"isbn": "9780000000011", "edition": "paper", "title": "Straße Atlas", "active": True, "key": "strasse atlas"},
+    {"isbn": "9780000000028", "edition": "paper", "title": "Zebra Tales", "active": False, "key": "zebra tales"},
+    {"isbn": "9780000000035", "edition": "paper", "title": "apple Almanac", "active": True, "key": "apple almanac"},
+]
+
+
+def pairs(rows):
+    return [(r["isbn"], r["edition"]) for r in rows]
+
+
+def test_schema_rows_and_order():
+    export = build_export()
+    assert list(export) == ["books"]
+    assert export["books"] == EXPECTED  # exact rows, exact order (A1)
+
+
+def test_inactive_included():
+    zebra = [r for r in build_export()["books"] if r["title"] == "Zebra Tales"]
+    assert len(zebra) == 1 and zebra[0]["active"] is False
+
+
+def test_key_length_differs_only_for_sharp_s():  # A2
+    diff = [r for r in build_export()["books"] if len(r["key"]) != len(r["title"])]
+    assert [r["title"] for r in diff] == ["Straße Atlas", "Straße Atlas"]
+    assert all(len(r["key"]) == 13 and len(r["title"]) == 12 for r in diff)
+
+
+def test_search_queries():
+    books = build_export()["books"]
+    for q in ("Straße", "stra", "STRASSE", "straße atlas"):
+        assert pairs(search(books, q)) == [("9780000000011", "large-print"), ("9780000000011", "paper")], q
+    assert pairs(search(books, "APP")) == [("9780000000035", "paper")]
+    assert pairs(search(books, "zeb")) == [("9780000000028", "paper")]
+    assert search(books, "xyz") == []
+
+
+def test_raw_lowercase_query_would_miss():  # A3: why the query must be casefolded
+    books = build_export()["books"]
+    assert not [r for r in books if r["key"].startswith("Straße".lower())]
+    assert len(search(books, "Straße")) == 2
+
+
+def test_cli_writes_valid_byte_stable_utf8(tmp_path):
+    a, b = tmp_path / "a.json", tmp_path / "b.json"
+    for out in (a, b):
+        subprocess.run([sys.executable, "catalog.py", "export", str(out)], cwd=ROOT, check=True)
+    assert a.read_bytes() == b.read_bytes()
+    raw = a.read_bytes()
+    assert raw.endswith(b"\n")
+    assert "Straße".encode("utf-8") in raw and b"\\u00df" not in raw
+    assert json.loads(raw) == {"books": EXPECTED}
