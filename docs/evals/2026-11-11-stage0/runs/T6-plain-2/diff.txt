+def build_export(rows):
+    """Build the ledger export: invoices sorted by id, lines in source order.
+
+    Each line is rounded to cents (ROUND_HALF_UP) and the invoice total is the
+    sum of the rounded lines, matching total(rows). Amounts are decimal strings.
+    """
+    grouped = {}
+    for row in rows:
+        grouped.setdefault(row["invoice"], []).append(row)
+    invoices = []
+    for invoice_id in sorted(grouped):
+        group = grouped[invoice_id]
+        lines = []
+        for row in group:
+            line = {"item": row["item"], "amount": str(line_total(row))}
+            if row.get("tax_class"):
+                line["tax_class"] = row["tax_class"]
+            lines.append(line)
+        invoices.append({"id": invoice_id, "total": str(total(group)), "lines": lines})
+    return {"invoices": invoices}

+def write_export(rows, destination):
+    import json
+    destination = Path(destination)
+    destination.parent.mkdir(parents=True, exist_ok=True)
+    with destination.open("w", encoding="utf-8", newline="\n") as stream:
+        json.dump(build_export(rows), stream, indent=2, sort_keys=True)
+        stream.write("\n")
+    return destination
