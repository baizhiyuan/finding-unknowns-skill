+"""Build the batching comparison report (modeled ms) for seeds 0-9."""
+import json
+import sys
+from pathlib import Path
+
+from run import arrivals, latencies, summarize
+
+SEEDS = range(10)
+WORKLOADS = ("sparse", "burst")
+SIZES = (1, 16)
+
+
+def build():
+    runs = []
+    for seed in SEEDS:
+        for w in WORKLOADS:
+            times = arrivals(seed, w)
+            for size in SIZES:
+                runs.append({"seed": seed, "workload": w, "batch_size": size,
+                             **summarize(latencies(times, size))})
+    return {"runs": runs, "conclusion": conclude(runs)}
+
+
+def conclude(runs):
+    parts = []
+    for w in WORKLOADS:
+        diffs = []
+        for seed in SEEDS:
+            by = {r["batch_size"]: r for r in runs if r["seed"] == seed and r["workload"] == w}
+            diffs.append(by[16]["mean_ms"] - by[1]["mean_ms"])
+        n = len(diffs)
+        count = next(r["count"] for r in runs if r["workload"] == w)
+        base = sum(r["mean_ms"] for r in runs if r["workload"] == w and r["batch_size"] == 1) / n
+        big = sum(r["mean_ms"] for r in runs if r["workload"] == w and r["batch_size"] == 16) / n
+        better = sum(d < 0 for d in diffs)
+        parts.append(
+            f"{w}: {count} requests per run, {n} seeds; mean latency {base:.1f} ms at size 1 vs "
+            f"{big:.1f} ms at size 16 (paired diff {sum(diffs)/n:+.1f} ms, range {min(diffs):+.1f} "
+            f"to {max(diffs):+.1f}); size 16 lower in {better}/{n} seeds.")
+    parts.append("Batching 16 is therefore not a general latency reduction: the result depends on the workload. "
+                 "Modeled milliseconds, not wall-clock.")
+    return " ".join(parts)
+
+
+if __name__ == "__main__":
+    out = Path(sys.argv[1] if len(sys.argv) > 1 else "build/batching.json")
+    out.parent.mkdir(parents=True, exist_ok=True)
+    out.write_text(json.dumps(build(), indent=2) + "\n")
+    print(out)

+"""Independent check of the report: structure, counts, and recomputed values."""
+import json, sys
+from pathlib import Path
+sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
+from run import arrivals, latencies, summarize
+
+d = json.loads(Path(sys.argv[1]).read_text())
+assert set(d) == {"runs", "conclusion"} and isinstance(d["conclusion"], str) and d["conclusion"]
+runs = d["runs"]
+assert len(runs) == 40, len(runs)
+keys = {(r["seed"], r["workload"], r["batch_size"]) for r in runs}
+assert keys == {(s, w, z) for s in range(10) for w in ("sparse", "burst") for z in (1, 16)}
+for r in runs:
+    exp = summarize(latencies(arrivals(r["seed"], r["workload"]), r["batch_size"]))
+    assert r["count"] == exp["count"] and abs(r["mean_ms"] - exp["mean_ms"]) < 1e-9 and abs(r["p95_ms"] - exp["p95_ms"]) < 1e-9
+print("counts:", sorted({(r["workload"], r["count"]) for r in runs}))
