+def build_export(rows):
+    """Build the ledger-compatible export: invoices grouped by id and sorted by id.
+
+    Source line order is preserved within each invoice. Every line is rounded
+    individually (see docs/accounting.md) and invoice totals sum those rounded
+    lines, so credits keep their sign and totals match total(rows).
+    """
+    grouped = {}
+    for row in rows:
+        grouped.setdefault(row["invoice"], []).append(row)
+    invoices = []
+    for invoice_id in sorted(grouped):
+        lines = []
+        for row in grouped[invoice_id]:
+            line = {"item": row["item"], "amount": str(line_total(row))}
+            if row.get("tax_class"):
+                line["tax_class"] = row["tax_class"]
+            lines.append(line)
+        invoices.append({"id": invoice_id, "total": str(total(grouped[invoice_id])), "lines": lines})
+    return {"invoices": invoices}
