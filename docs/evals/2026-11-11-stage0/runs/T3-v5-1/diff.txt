+def report(seeds=range(10)):
+    runs = []
+    for seed in seeds:
+        for workload in ("sparse", "burst"):
+            times = arrivals(seed, workload)
+            for size in (1, 16):
+                runs.append({"seed": seed, "workload": workload, "batch_size": size,
+                             **summarize(latencies(times, size))})
+    return {"runs": runs, "conclusion": conclude(runs)}
+

+def conclude(runs):
+    def mean(workload, size):
+        rows = [r["mean_ms"] for r in runs if r["workload"] == workload and r["batch_size"] == size]
+        return sum(rows) / len(rows)
+    def counts(workload):
+        return sorted({r["count"] for r in runs if r["workload"] == workload})
+    parts = []
+    for workload in ("sparse", "burst"):
+        one, sixteen = mean(workload, 1), mean(workload, 16)
