+"""Build the prefix-search export (build/search.json) from data/books.json."""
+import json
+from pathlib import Path
+import sys
+
+from catalog import load, search_key
+
+
+def build_export(path="data/books.json"):
+    rows = [{"isbn": r["isbn"], "edition": r["edition"], "title": r["title"],
+             "active": r["active"], "key": search_key(r["title"])} for r in load(path)]
+    rows.sort(key=lambda r: (r["title"], r["isbn"], r["edition"]))
+    return {"books": rows}
+
+
+def write_export(destination="build/search.json", path="data/books.json"):
+    destination = Path(destination)
+    destination.parent.mkdir(parents=True, exist_ok=True)
+    text = json.dumps(build_export(path), indent=2, ensure_ascii=False) + "\n"
+    destination.write_text(text, encoding="utf-8")
+    return destination
+
+
+def search(prefix, path="data/books.json"):
+    """Rows whose key starts with the case-folded prefix; inactive stock included."""
+    folded = search_key(prefix)
+    return [r for r in build_export(path)["books"] if r["key"].startswith(folded)]
+
+
+if __name__ == "__main__":
+    print(write_export(*sys.argv[1:2]))

+import json
+import subprocess
+import sys
+
+from export import build_export, write_export, search
+
+EXPECTED_ORDER = [
+    ("9780000000011", "large-print"),
+    ("9780000000011", "paper"),
+    ("9780000000028", "paper"),
+    ("9780000000035", "paper"),
+]
+
+
+def books():
+    return build_export()["books"]
+
+
+def test_all_rows_unique_identity_inactive_included():
+    rows = books()
+    assert len(rows) == 4
+    assert {(r["isbn"], r["edition"]) for r in rows} == set(EXPECTED_ORDER)
+    assert {r["isbn"]: r["active"] for r in rows}["9780000000028"] is False
+
+
+def test_key_is_casefold_and_title_unchanged():
+    rows = {(r["isbn"], r["edition"]): r for r in books()}
+    r = rows[("9780000000011", "paper")]
+    assert r["title"] == "Straße Atlas"
+    assert r["key"] == "strasse atlas"
+    assert len(r["title"]) == 12 and len(r["key"]) == 13
+    assert rows[("9780000000035", "paper")]["title"] == "apple Almanac"
+    assert rows[("9780000000035", "paper")]["key"] == "apple almanac"
+
+
+def test_order():
+    assert [(r["isbn"], r["edition"]) for r in books()] == EXPECTED_ORDER
+
+
+def test_schema():
+    data = build_export()
+    assert list(data) == ["books"]
+    for r in books():
+        assert set(r) == {"isbn", "edition", "title", "active", "key"}
+        assert all(isinstance(r[k], str) for k in ("isbn", "edition", "title", "key"))
+        assert isinstance(r["active"], bool)
+
+
+def ids(rows):
+    return [(r["isbn"], r["edition"]) for r in rows]
+
+
+def test_search_queries():
+    both = [("9780000000011", "large-print"), ("9780000000011", "paper")]
+    assert ids(search("str")) == both
+    assert ids(search("STRASSE")) == both
+    assert ids(search("straß")) == both
+    assert ids(search("zeb")) == [("9780000000028", "paper")]
+    assert search("zeb")[0]["title"] == "Zebra Tales"
+    assert search("nomatch") == []
+
+
+def test_cli_prints_full_original_title():
+    out = subprocess.run([sys.executable, "catalog.py", "search", "STRASSE"],
+                         capture_output=True, text=True, check=True, encoding="utf-8").stdout
+    assert out.count("Straße Atlas") == 2
+
+
+def test_written_file_matches_builder(tmp_path):
+    target = tmp_path / "build" / "search.json"
+    write_export(target)
+    assert json.loads(target.read_text(encoding="utf-8")) == build_export()
